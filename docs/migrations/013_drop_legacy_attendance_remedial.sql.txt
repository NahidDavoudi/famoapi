-- ============================================================================
-- 013_drop_legacy_attendance_remedial
-- ============================================================================
-- OPERATOR-ONLY REFERENCE. DO NOT RUN AUTOMATICALLY.
--
-- This script drops legacy tables that are no longer used by the application.
-- The `remedial_*` and `session_*` tables do not exist in this deployment (the
-- Remedial module was dead code and has been removed), and the `attendance_*`
-- tables were superseded by the daily attendance module.
--
-- It MUST be run MANUALLY by an operator, and ONLY AFTER taking a verified
-- backup. The `.txt` extension is deliberate: the auto-running migrator only
-- executes `database/migrations/*.sql`, so this file is never run automatically.
--
-- Usage (review, back up, then execute):
--   mysql -u <user> -p <database> < 013_drop_legacy_attendance_remedial.sql.txt
-- ============================================================================


-- ----------------------------------------------------------------------------
-- STEP 1: BACK UP THE LEGACY `attendance_*` TABLES (run this first!)
-- ----------------------------------------------------------------------------
-- Uncomment and run the following command manually, then confirm the dump file
-- exists and is non-empty before proceeding to STEP 2.
--
-- mysqldump -u <user> -p <database> \
--   attendance_records \
--   attendance_session_students \
--   attendance_sessions \
--   attendance_classes \
--   attendance_archive \
--   > backup_attendance_legacy_$(date +%Y%m%d_%H%M%S).sql
--
-- On Windows (PowerShell), a timestamped backup looks like:
--   mysqldump -u <user> -p <database> attendance_records attendance_session_students attendance_sessions attendance_classes attendance_archive > backup_attendance_legacy_$(Get-Date -Format yyyyMMdd_HHmmss).sql


-- ----------------------------------------------------------------------------
-- STEP 2: DROP LEGACY TABLES
-- Run only after STEP 1 backup is verified.
-- ----------------------------------------------------------------------------
SET FOREIGN_KEY_CHECKS = 0;

DROP TABLE IF EXISTS attendance_records;
DROP TABLE IF EXISTS attendance_session_students;
DROP TABLE IF EXISTS attendance_sessions;
DROP TABLE IF EXISTS attendance_classes;
DROP TABLE IF EXISTS attendance_archive;
DROP TABLE IF EXISTS session_student_times;
DROP TABLE IF EXISTS remedial_attendance;
DROP TABLE IF EXISTS session_students;
DROP TABLE IF EXISTS remedial_classes;
DROP TABLE IF EXISTS remedial_sessions;

SET FOREIGN_KEY_CHECKS = 1;
